const SERVER_ADDRESS = "PrisonSMPMC.aternos.me:19912"; async function checkServerStatus() { const statusText = document.getElementById("statusText"); const statusDetails = document.getElementById("statusDetails"); if (!statusText || !statusDetails) { console.error("Server status elements were not found."); return; } statusText.textContent = "CHECKING..."; statusDetails.textContent = "Checking the Minecraft server..."; try { const url = "https://api.mcsrvstat.us/3/" + encodeURIComponent(SERVER_ADDRESS) + "?_=" + Date.now(); const response = await fetch(url, { cache: "no-store" }); if (!response.ok) { throw new Error("Status service returned HTTP " + response.status); } const data = await response.json(); if (data.online === true) { statusText.textContent = "ONLINE"; const playersOnline = data.players && typeof data.players.online === "number" ? data.players.online : null; const playerText = playersOnline === null ? "Player count unavailable." : playersOnline + (playersOnline === 1 ? " player online." : " players online."); const versionText = data.version ? " Version: " + data.version + "." : ""; statusDetails.textContent = playerText + versionText; } else if (data.online === false) { statusText.textContent = "OFFLINE"; statusDetails.textContent = "The status service reports the server as offline. If people are playing, this result may be incorrect."; } else { statusText.textContent = "STATUS UNKNOWN"; statusDetails.textContent = "The status service did not return a clear server status."; } } catch (error) { console.error("Server status check failed:", error); statusText.textContent = "STATUS UNAVAILABLE"; statusDetails.textContent = "Could not check the server right now. Try again shortly."; } } // Check immediately and then every 60 seconds. checkServerStatus(); setInterval(checkServerStatus, 60000);
